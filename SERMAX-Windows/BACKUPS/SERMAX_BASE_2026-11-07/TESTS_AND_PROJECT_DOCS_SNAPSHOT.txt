import unittest

from sermax_widget_state import ESTADOS_VISUAIS, WidgetStateController


class WidgetStateControllerTests(unittest.TestCase):
    def setUp(self):
        self.controller = WidgetStateController()

    def test_initial_state_is_idle(self):
        self.assertEqual(self.controller.state.name, "idle")

    def test_known_events_map_to_closed_visual_states(self):
        for event in ("idle", "attention", "processing", "completed", "error"):
            with self.subTest(event=event):
                state = self.controller.set_event(event)
                self.assertEqual(state.name, event)
                self.assertIn(state.name, ESTADOS_VISUAIS)

    def test_unknown_or_non_string_event_fails_closed(self):
        self.assertEqual(self.controller.set_event("run_shell").name, "error")
        self.assertEqual(self.controller.set_event(None).name, "error")

    def test_neural_output_is_restricted_to_visual_state_allowlist(self):
        self.assertEqual(self.controller.accept_neural_output(" attention ").name, "attention")
        self.assertEqual(self.controller.accept_neural_output("run_command").name, "error")
        self.assertEqual(self.controller.accept_neural_output({"action": "shutdown"}).name, "error")

    def test_controller_does_not_change_state_outside_explicit_calls(self):
        self.controller.set_event("processing")
        self.assertEqual(self.controller.state.name, "processing")


if __name__ == "__main__":
    unittest.main()

import tempfile
import unittest
import sys
from unittest.mock import patch

import sermax_workspace
from pathlib import Path

from sermax_workspace import WorkspaceAdapter, TestRunner, WorkspaceError


class WorkspaceAdapterTests(unittest.TestCase):
    def test_read_write_and_backup(self):
        with tempfile.TemporaryDirectory() as folder:
            root = Path(folder)
            (root / "src").mkdir()
            (root / "src" / "sample.py").write_text("old", encoding="utf-8")
            ws = WorkspaceAdapter(root)
            self.assertEqual(ws.read_text("src/sample.py"), "old")
            ws.write_text("src/sample.py", "new")
            self.assertEqual(ws.read_text("src/sample.py"), "new")
            self.assertEqual(
                (root / ".sermax_dev_backups" / "src" / "sample.py").read_text(),
                "old",
            )

    def test_blocks_protected_and_traversal_paths(self):
        with tempfile.TemporaryDirectory() as folder:
            ws = WorkspaceAdapter(folder)
            for name in ("../outside.txt", "..\\outside.txt", "memoria_sermax.json",
                         "memoria_neural_pestinha.json", ".git/config"):
                with self.subTest(name=name), self.assertRaises(WorkspaceError):
                    ws.read_text(name)

    def test_blocks_symlink_escape(self):
        with tempfile.TemporaryDirectory() as folder, tempfile.TemporaryDirectory() as outside:
            root = Path(folder)
            (root / "escape").symlink_to(Path(outside), target_is_directory=True)
            ws = WorkspaceAdapter(root)
            with self.assertRaises(WorkspaceError):
                ws.write_text("escape/file.txt", "blocked")

    def test_test_runner_allowlist(self):
        with tempfile.TemporaryDirectory() as folder:
            root = Path(folder)
            tests_dir = root / "tests"
            tests_dir.mkdir()
            (tests_dir / "test_smoke.py").write_text(
                "import unittest\n"
                "class SmokeTests(unittest.TestCase):\n"
                "    def test_runner_executes_discovered_tests(self):\n"
                "        self.assertTrue(True)\n",
                encoding="utf-8",
            )
            ws = WorkspaceAdapter(root)
            runner = TestRunner(ws)
            with self.assertRaises(WorkspaceError):
                runner.run("arbitrary-shell")
            result = runner.run("unittest")
            self.assertEqual(result["status"], "passed", result["output"])
            self.assertIn("Ran 1 test", result["output"])


    def test_test_runner_enforces_output_limit_during_execution(self):
        with tempfile.TemporaryDirectory() as folder:
            ws = WorkspaceAdapter(folder)
            runner = TestRunner(ws, timeout_seconds=10)
            original = sermax_workspace.ALLOWED_TESTS["compileall"]
            try:
                sermax_workspace.ALLOWED_TESTS["compileall"] = (
                    sys.executable, "-c",
                    "import sys; sys.stdout.write('x' * 200000); sys.stdout.flush()",
                )
                result = runner.run("compileall")
                self.assertEqual(result["status"], "output_limit")
                self.assertTrue(result["truncated"])
            finally:
                sermax_workspace.ALLOWED_TESTS["compileall"] = original

    def test_backup_symlink_is_rejected(self):
        with tempfile.TemporaryDirectory() as folder, tempfile.TemporaryDirectory() as outside:
            root = Path(folder)
            (root / ".sermax_dev_backups").symlink_to(Path(outside), target_is_directory=True)
            (root / "file.txt").write_text("old", encoding="utf-8")
            ws = WorkspaceAdapter(root)
            with self.assertRaises(WorkspaceError):
                ws.write_text("file.txt", "new")


    def test_isolated_workspace_copy_excludes_credentials_and_build_artifacts(self):
        with tempfile.TemporaryDirectory() as folder:
            root = Path(folder) / "workspace"
            root.mkdir()
            (root / "module.py").write_text("VALUE = 1\n", encoding="utf-8")
            (root / ".venv-jarvis").mkdir()
            (root / ".venv-jarvis" / "secretless-runtime.txt").write_text("runtime")
            (root / ".deps" / "pestinha").mkdir(parents=True)
            (root / ".deps" / "pestinha" / "brain.py").write_text("VALUE = 2")
            (root / ".sermax_dev_backups").mkdir()
            (root / ".sermax_dev_backups" / "old.py").write_text("OLD = 1")
            workspace = WorkspaceAdapter(root)
            runner = TestRunner(workspace, isolate_as_nobody=True)
            with tempfile.TemporaryDirectory() as temporary:
                with patch("sermax_workspace.subprocess.run") as mocked:
                    sandbox = runner._prepare_isolated_workspace(temporary)
                self.assertTrue((sandbox / "module.py").is_file())
                self.assertTrue((sandbox / ".deps" / "pestinha" / "brain.py").is_file())
                self.assertFalse((sandbox / ".venv-jarvis").exists())
                self.assertFalse((sandbox / ".sermax_dev_backups").exists())
                self.assertEqual(mocked.call_count, 2)

    def test_windows_install_path_is_rejected_by_path_guard(self):
        self.assertTrue(sermax_workspace._is_inside_windows_install(r"C:\\SERMAX"))
        self.assertTrue(sermax_workspace._is_inside_windows_install(r"C:\\SERMAX\\workspace"))
        self.assertFalse(sermax_workspace._is_inside_windows_install(r"C:\\SERMAX_DEV"))

if __name__ == "__main__":
    unittest.main()

# SERMAX: equipe única de desenvolvimento coordenada pelo MAX

**Diretriz vigente:** uma única equipe de agentes desenvolve, programa, analisa, pesquisa, testa, corrige e evolui o código-fonte do SERMAX. A equipe opera no ambiente de desenvolvimento, fora do executável instalado. O SERMAX instalado recebe atualizações somente após implementação e validação.

## Papéis sob demanda

- **Programador:** propõe e implementa alterações em workspace isolado.
- **Analista:** inspeciona código, falhas, dependências e resultados.
- **Testador:** executa testes permitidos, reproduz falhas e verifica correções.
- **Pesquisador:** consulta documentação, APIs, licenças e fontes técnicas.

MAX coordena a fila e aciona somente os papéis necessários. Os papéis compartilham um único backend de modelo, trabalham sequencialmente por padrão e não mantêm modelos residentes separados. Paralelismo fica desativado até haver evidência de benefício sem pressão de RAM/CPU.

## Fora do executável SERMAX

A equipe de desenvolvimento não precisa ser empacotada no `SERMAX.exe`. Ela roda no ambiente de desenvolvimento, com acesso controlado à branch e aos ambientes de teste. O executável instalado mantém o serviço normal de assistência. As melhorias chegam à instalação por um processo de atualização aprovado, com backup e rollback.

## Componentes existentes reaproveitados

| Componente | Uso nesta diretriz |
|---|---|
| `sermax_planejador.py` | Reaproveitar conceitos de planejamento; hoje não altera arquivos. |
| `sermax_executor.py` | Reaproveitar validação de ações locais; hoje não edita código nem executa testes. |
| `sermax_tarefas.py` | Reaproveitar execução sequencial; hoje não mantém fila persistente. |
| OpenJarvis | Motor opcional/adaptador; worker atual usa agente `simple`, sem ferramentas/contexto próprios do SDK. |
| Pestinha | Subsistema comportamental e visual; memória neural continua separada. Não é agente programador. |

## Estado da implementação

Foi iniciado `sermax_dev_team.py`, supervisor externo com fila SQLite persistente, ativação sequencial dos quatro papéis e callback para um único modelo compartilhado. O worker em segundo plano processa uma tarefa por vez.

**Limite atual explícito:** o supervisor ainda não está conectado a um backend de modelo, GitHub, editor de workspace ou executor de testes. As respostas dos papéis são análises/propostas; este módulo não grava código, não executa testes e não faz commits. Portanto, não representa autonomia de desenvolvimento concluída.

## Próximas integrações necessárias

1. Adaptador de modelo compartilhado, com Jarvis opcional e fallback Ollama, sem carregar um modelo por agente.
2. Workspace isolado ligado à branch autorizada, com verificação de branch/HEAD antes de qualquer escrita.
3. Ferramentas restritas de leitura e edição de arquivos, com diff, limites de caminho/tamanho e rollback.
