import json
import os
import tempfile
import unittest

import sermax_core as core

        self.path_original = core.ARQUIVO_MEMORIA
        self.historico_original = list(core.historico)
        core.ARQUIVO_MEMORIA = os.path.join(self.tmp.name, "memoria_sermax.json")
        core.historico[:] = []

    def tearDown(self):
        core.ARQUIVO_MEMORIA = self.path_original
        core.historico[:] = self.historico_original
        self.tmp.cleanup()

    def test_memoria_pessoal_valida(self):
        self.assertTrue(core.eh_memoria_candidata("Meu nome é Fabiano"))
        self.assertTrue(core.eh_memoria_candidata("Meu carro é um Fiat Siena 2008"))

    def test_nao_grava_pergunta_ou_comando(self):
        self.assertFalse(core.eh_memoria_candidata("Qual óleo uso?"))
        self.assertFalse(core.eh_memoria_candidata("/memoria"))
        self.assertFalse(core.eh_memoria_candidata("continua"))

    def test_persistencia_e_filtro_de_resposta_assistente(self):
        core.historico.extend([
            {"role": "user", "content": "Meu nome é Fabiano"},
            {"role": "assistant", "content": "Seu nome é Fabiano"},
            {"role": "user", "content": "Meu carro é um Fiat Siena 2008"},
        ])
        core.salvar_memoria()
        with open(core.ARQUIVO_MEMORIA, encoding="utf-8") as f:
            dados = json.load(f)
        self.assertEqual([x["content"] for x in dados], [
            "Meu nome é Fabiano", "Meu carro é um Fiat Siena 2008"
        ])
        self.assertTrue(os.path.isfile(core.ARQUIVO_MEMORIA))

    def test_memoria_recarrega_apos_reinicializacao(self):
        core.historico.append({"role": "user", "content": "Meu carro é um Fiat Siena 2008"})
        core.salvar_memoria()
        core.historico[:] = []
        recarregada = core.carregar_memoria()
        self.assertEqual(len(recarregada), 1)
        self.assertEqual(recarregada[0]["content"], "Meu carro é um Fiat Siena 2008")

    def test_falha_ao_substituir_preserva_memoria_anterior(self):
        conteudo_anterior = '[{"role": "user", "content": "Meu nome é Fabiano"}]'
        with open(core.ARQUIVO_MEMORIA, "w", encoding="utf-8") as f:
            f.write(conteudo_anterior)
        core.historico[:] = [
            {"role": "user", "content": "Meu carro é um Fiat Siena 2008"}
        ]

        from unittest.mock import patch
        with patch("sermax_core.os.replace", side_effect=OSError("falha simulada")):
            with self.assertRaises(OSError):
                core.salvar_memoria()

        with open(core.ARQUIVO_MEMORIA, encoding="utf-8") as f:
            self.assertEqual(f.read(), conteudo_anterior)
        temporarios = [
            nome for nome in os.listdir(self.tmp.name)
            if nome.startswith("memoria_sermax.json.tmp_")
        ]
        self.assertEqual(temporarios, [])

    def test_memoria_corrompida_preserva_backup(self):
        with open(core.ARQUIVO_MEMORIA, "w", encoding="utf-8") as f:

        recarregada = core.carregar_memoria()

        self.assertEqual(recarregada, [])

import unittest
from pathlib import Path

from sermax_workspace import WorkspaceAdapter
from sermax_dev_proposal import (
    ProposalError, apply_proposal, apply_and_test_proposal, parse_proposal, preview_proposal,

class DevelopmentProposalTests(unittest.TestCase):
    def test_parse_preview_and_apply_structured_proposal(self):
        with tempfile.TemporaryDirectory() as folder:
            root = Path(folder)
            (root / "src").mkdir()
            (root / "src" / "module.py").write_text("old = 1\n", encoding="utf-8")
            workspace = WorkspaceAdapter(root)
            raw = json.dumps({
                "summary": "Atualiza módulo de teste",
                "files": [{"path": "src/module.py", "content": "new = 2\n"}],
                "tests": ["unittest", "compileall"],
            })
            proposal = parse_proposal(raw)
            self.assertIn("-old = 1", preview_proposal(workspace, proposal)[0]["diff"])
            self.assertEqual(apply_proposal(workspace, proposal), ["src/module.py"])
            self.assertEqual(workspace.read_text("src/module.py"), "new = 2\n")
            self.assertEqual(
                (root / ".sermax_dev_backups" / "src" / "module.py").read_text(encoding="utf-8"),
                "old = 1\n",
            )

    def test_rejects_invalid_json_schema_and_test_command(self):
        invalid = (
            "not-json",
            "{}",
            json.dumps({
                "summary": "x", "files": [{"path": "a.py", "content": "x"}],
                "tests": ["powershell"],
            }),
            json.dumps({
                "summary": "x", "files": [{"path": "a.py", "content": "x"}],
                "tests": ["compileall"],
            }),
        )
        for raw in invalid:
            with self.subTest(raw=raw), self.assertRaises(ProposalError):
                parse_proposal(raw)

    def test_rejects_duplicate_and_protected_paths(self):
        duplicate = json.dumps({
            "summary": "x",
            "files": [
                {"path": "src/a.py", "content": "1"},
                {"path": "SRC/A.PY", "content": "2"},
            ],
            "tests": ["unittest"],
        })
        with self.assertRaises(ProposalError):
            parse_proposal(duplicate)
        protected_paths = (
            ".github/workflows/ci.yml",
            "sermax_github_agent_runner.py",
            "sermax_groq_backend.py",
            "sermax_dev_team.py",
            "memoria_sermax.json",
            "settings.json",
            "setup.ps1",
            "../outside.py",
            "SERMAX_terminal/sermax_terminal.py",
        )
        for protected in protected_paths:
            with self.subTest(path=protected), self.assertRaises(ProposalError):
                parse_proposal(json.dumps({
                    "summary": "x",
                    "files": [{"path": protected, "content": "x"}],
                    "tests": ["unittest"],
                }))

    def test_apply_and_test_rolls_back_when_tests_fail(self):
        class FailedRunner:
            def run(self, suite):
                return {"suite": suite, "status": "failed", "returncode": 1, "output": "failure"}

        with tempfile.TemporaryDirectory() as folder:
            root = Path(folder)
            (root / "module.py").write_text("ORIGINAL = True\n", encoding="utf-8")

            proposal = parse_proposal(json.dumps({

                    {"path": "new_module.py", "content": "NEW = True\n"},
                ],
                "tests": ["unittest"],
            }))
            with self.assertRaises(RuntimeError):
                apply_and_test_proposal(workspace, proposal, FailedRunner())
            self.assertEqual(workspace.read_text("module.py"), "ORIGINAL = True\n")
            self.assertFalse((root / "new_module.py").exists())

        with tempfile.TemporaryDirectory() as folder:
            calls = []
            def model_call(role, prompt):
                calls.append((role, prompt))
                return f"resultado-{role}"

            team = DevelopmentTeam(Path(folder) / "team.sqlite3", model_call)
            task_id = team.enqueue(
                "Corrigir teste", "Falha reproduzida",
                roles=("analista", "programador", "testador"),
            )
            self.assertEqual(team.run_next(), task_id)
            task = team.get_task(task_id)
            self.assertEqual(task["status"], "completed")
            self.assertEqual([role for role, _ in calls],
                             ["analista", "programador", "testador"])
            self.assertTrue(all("Corrigir teste" in prompt for _, prompt in calls))
            self.assertEqual(len(team.events(task_id)), 6)

    def test_queue_survives_restart(self):
        with tempfile.TemporaryDirectory() as folder:
            path = Path(folder) / "team.sqlite3"
            first = DevelopmentTeam(path, lambda role, prompt: "ok")
            task_id = first.enqueue("Inspecionar código")
            second = DevelopmentTeam(path, lambda role, prompt: "ok")
            self.assertEqual(second.get_task(task_id)["status"], "queued")

    def test_rejects_unknown_roles(self):

            team = DevelopmentTeam(Path(folder) / "team.sqlite3", lambda r, p: "ok")

    def _wav_pcm_minimo(self, caminho):
        fmt = struct.pack("<HHIIHH", 1, 1, 16000, 32000, 2, 16)
        dados = b"\x00\x00\x01\x00"
        chunks = b"fmt " + struct.pack("<I", len(fmt)) + fmt
        chunks += b"data" + struct.pack("<I", len(dados)) + dados
        conteudo = b"WAVE" + chunks
        caminho.write_bytes(b"RIFF" + struct.pack("<I", len(conteudo)) + conteudo)
