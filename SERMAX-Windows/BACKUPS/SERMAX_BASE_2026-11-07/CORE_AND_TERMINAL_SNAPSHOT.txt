def normalizar(texto):

_OPERADORES_BINARIOS={ast.Add:operator.add,ast.Sub:operator.sub,ast.Mult:operator.mul,ast.Div:operator.truediv,ast.FloorDiv:operator.floordiv,ast.Mod:operator.mod,ast.Pow:operator.pow}
_OPERADORES_UNARIOS={ast.UAdd:operator.pos,ast.USub:operator.neg}
def _avaliar_conta(no):
    if isinstance(no,ast.Expression): return _avaliar_conta(no.body)
    if isinstance(no,ast.Constant) and isinstance(no.value,(int,float)) and not isinstance(no.value,bool): valor=no.value
    elif isinstance(no,ast.BinOp) and type(no.op) in _OPERADORES_BINARIOS:
        esq,dir=_avaliar_conta(no.left),_avaliar_conta(no.right)
        if isinstance(no.op,ast.Pow) and (abs(dir)>10 or abs(esq)>1000000): raise ValueError("Potência fora do limite seguro.")
        valor=_OPERADORES_BINARIOS[type(no.op)](esq,dir)
    elif isinstance(no,ast.UnaryOp) and type(no.op) in _OPERADORES_UNARIOS: valor=_OPERADORES_UNARIOS[type(no.op)](_avaliar_conta(no.operand))
    else: raise ValueError("Expressão não permitida.")
    if not math.isfinite(float(valor)) or abs(valor)>1e15: raise ValueError("Resultado fora do limite seguro.")
    return valor

def _formatar_resultado(valor):
    if isinstance(valor,float) and valor.is_integer(): valor=int(valor)
    if isinstance(valor,int): return f"{valor:,}".replace(",",".")
    return f"{valor:,.8f}".rstrip("0").rstrip(".").replace(",","X").replace(".",",").replace("X",".")

def responder_calculo_local(mensagem):
    texto=str(mensagem).strip().lower().replace("×","*").replace("÷","/")
    texto=re.sub(r"(?<![\w.])(\d+(?:[.,]\d+)?)\s*%\s+de\s+([+-]?\d+(?:[.,]\d+)?)",r"(\1/100*\2)",texto)
    texto=re.sub(r"(?<=\d),(?=\d)",".",texto)
    texto=re.sub(r"(?<=[0-9)])\s*[xX]\s*(?=[0-9(])","*",texto)
    texto=re.sub(r"^(?:por favor[, ]*)?(?:calcule|calcular|quanto é|quanto e|resolva|resultado de|faz a conta|faça a conta|faca a conta)\s*","",texto)
    texto=re.sub(r"\s+","",texto)
    if not texto or len(texto)>100 or not re.fullmatch(r"[0-9.+*/()%\-]+",texto) or not re.search(r"[+*/%\-]",texto): return None
    try: return f"Resultado: {_formatar_resultado(_avaliar_conta(ast.parse(texto,mode='eval')))}"
    except ZeroDivisionError: return "Não consigo dividir por zero."
    except (SyntaxError,ValueError,TypeError,OverflowError): return None

def _perguntar_impl(mensagem,receber_parte=None):
    global ultimo_diagnostico_ollama
    mensagem=str(mensagem).strip()
    if not mensagem: return ""
    resposta=responder_comando_local(mensagem)
    if resposta is None: resposta=responder_calculo_local(mensagem)
    if resposta is not None:
        sessao.extend([{"role":"user","content":mensagem},{"role":"assistant","content":resposta}])
        if receber_parte: receber_parte(resposta)
        return resposta
    sessao.append({"role":"user","content":mensagem}); registrar_memoria(mensagem)
    ferramenta=ferramenta_windows.responder_consulta(mensagem)
    if ferramenta is not None:
        sessao.append({"role":"assistant","content":ferramenta})
        if receber_parte: receber_parte(ferramenta)
        return ferramenta
    if MOTOR_RESPOSTA == "jarvis":
        # O SERMAX fornece identidade, memória recuperada e contexto da sessão.
        # OpenJarvis permanece sem ferramentas e sem memória própria.
        mensagens = montar_contexto(mensagem)
        prompt = "\n\n".join(
            f"{item['role'].upper()}: {item['content']}" for item in mensagens
        )
        prompt = prompt[-3900:]
        resposta = sermax_jarvis_adapter.JarvisAdapter(model=MODELO).responder(prompt)
        sessao.append({"role":"assistant","content":resposta})
        if receber_parte: receber_parte(resposta)
        return resposta
    ultimo_diagnostico_ollama={}
    dados={"model":MODELO,"messages":montar_contexto(mensagem),"stream":True,"keep_alive":KEEP_ALIVE,"options":OPCOES_MODELO}
    pedido=urllib.request.Request(OLLAMA_URL,data=json.dumps(dados,ensure_ascii=False).encode("utf-8"),headers={"Content-Type":"application/json"},method="POST")
    resposta=""
    try:
        with urllib.request.urlopen(pedido,timeout=300) as retorno:
            for linha in retorno:
                if not linha: continue
                try: obj=json.loads(linha.decode("utf-8"))
                except json.JSONDecodeError: continue
                trecho=obj.get("message",{}).get("content","")
                if trecho:
                    resposta+=trecho
                    if receber_parte: receber_parte(resposta)
                if obj.get("done"):
                    campos=("load_duration","prompt_eval_duration","eval_duration","total_duration","prompt_eval_count","eval_count")
                    ultimo_diagnostico_ollama={k:obj[k] for k in campos if k in obj}; break
    except urllib.error.URLError as e:
        raise RuntimeError("Não foi possível conectar ao Ollama. Verifique se o Ollama está aberto.") from e
    sessao.append({"role":"assistant","content":resposta})
    return resposta.strip()


def perguntar(mensagem, receber_parte=None):
    """Envolve cada solicitação com eventos neurais isolados e fallback seguro."""
    atualizar_estado_widget("processing")
    try:
        resposta = _perguntar_impl(mensagem, receber_parte=receber_parte)
        atualizar_estado_widget("completed")
        return resposta
    except Exception:
        atualizar_estado_widget("error")
        raise
    finally:
        # A persistência neural nunca escreve em memoria_sermax.json.
        encerrar_widget()

import os

def resolver_caminhos(frozen, executable, arquivo_fonte, meipass=None):
    """Resolve a pasta persistente da instalação e a pasta da interface empacotada."""
    root = os.path.dirname(os.path.abspath(executable if frozen else arquivo_fonte))
    bundle_root = meipass if frozen and meipass else root
    terminal_dir = os.path.join(bundle_root, "SERMAX_terminal")
    terminal_script = os.path.join(terminal_dir, "sermax_terminal.py")
    return root, bundle_root, terminal_dir, terminal_script


def main():
    frozen = bool(getattr(sys, "frozen", False))
    root, bundle_root, terminal_dir, terminal_script = resolver_caminhos(
        frozen,
        sys.executable,
        __file__,
        getattr(sys, "_MEIPASS", None),
    )
    os.environ["SERMAX_ROOT"] = root
    if not os.path.isfile(terminal_script):
        print("SERMAX: não encontrei a interface terminal em:")
        print(terminal_script)
        input("Pressione Enter para fechar.")
        return
    os.chdir(root)
    for path in (root, bundle_root, terminal_dir):
        if path not in sys.path:
            sys.path.insert(0, path)
    runpy.run_path(terminal_script, run_name="__main__")


if __name__ == "__main__":
    main()

def reduzir_rosto(texto, linhas_alvo=25, colunas_alvo=66):
    linhas = texto.splitlines()

    while linhas and not linhas[0].strip():
        linhas.pop(0)

    while linhas and not linhas[-1].strip():
        linhas.pop()

    if not linhas:
        return []

    altura = len(linhas)
    largura = max(len(linha) for linha in linhas)

    resultado = []

    for y in range(linhas_alvo):
        oy = min(altura - 1, int(y * altura / linhas_alvo))
        linha = linhas[oy].ljust(largura)

        resultado.append(
            "".join(
                linha[
                    min(
                        largura - 1,
                        int(x * largura / colunas_alvo)
                    )
                ]
                for x in range(colunas_alvo)
            ).rstrip()
        )

    return resultado

def cortar(texto, largura):
    texto = str(texto)

    if len(texto) <= largura:
        return texto

    return texto[:max(0, largura - 1)] + "…"


def quebrar(texto, largura):
    resultado = []

    for bloco in str(texto).replace("\r", "").split("\n"):

        if not bloco:
            resultado.append("")
            continue

        palavras = bloco.split()
        atual = ""

        for palavra in palavras:

            if len(palavra) > largura:

                if atual:
                    resultado.append(atual)
                    atual = ""

                for i in range(0, len(palavra), largura):
                    resultado.append(
                        palavra[i:i + largura]
                    )

            elif not atual:
                atual = palavra

            elif len(atual) + 1 + len(palavra) <= largura:
                atual += " " + palavra

            else:
                resultado.append(atual)
                atual = palavra

        if atual:
            resultado.append(atual)

    return resultado


def preparar_terminal():

    if os.name == "nt":

        os.system(
            f"mode con: cols={COLS} lines={ROWS} >nul 2>&1"
        )

        try:
            import ctypes

            k32 = ctypes.windll.kernel32
            h = k32.GetStdHandle(-11)

            mode = ctypes.c_uint32()

            if k32.GetConsoleMode(
                h,
                ctypes.byref(mode)
            ):
                k32.SetConsoleMode(
                    h,
                    mode.value | 0x0004
                )

        except Exception:
            pass

    os.system("title SERMAX - Terminal de IA")

def render(conversa):

    tela = []

    tela.append(
        " SERMAX - Terminal de IA".ljust(COLS)
    )

    tela.append("-" * COLS)

    rosto = ROSTO_REDUZIDO[:TOP_ROWS - 2]

    p = painel()

    for i in range(TOP_ROWS - 2):

        esquerda = cortar(
            rosto[i] if i < len(rosto) else "",
            FACE_COLS
        ).ljust(FACE_COLS)

        direita = (
            p[i]
            if i < len(p)
            else "|" +
            " " * (PANEL_COLS - 2) +
            "|"
        )

        tela.append(
            (
                " " +
                esquerda +
                " " +
                direita
            ).ljust(COLS)
        )

    tela.append(
        "+" +
        "-" * (COLS - 2) +
        "+"
    )

    tela.append(
        "|" +
        " CONVERSA".ljust(COLS - 2) +
        "|"
    )

    tela.append(
        "|" +
        "-" * (COLS - 2) +
        "|"
    )

    largura = COLS - 5
    capacidade = CHAT_ROWS - 1

    chat_reverso = []

    for item in reversed(conversa):

        prefixo = (
            "Você: "
            if item["role"] == "user"
            else "SERMAX: "
        )

        partes = quebrar(
            item["content"],
            largura - len(prefixo)
        ) or [""]

        linhas_item = [
            prefixo + partes[0]
        ]

        linhas_item.extend(
            " " * len(prefixo) + x
            for x in partes[1:]
        )

        for linha in reversed(linhas_item):

            chat_reverso.append(linha)

            if len(chat_reverso) >= capacidade:
                break

        if len(chat_reverso) >= capacidade:
            break

    chat = list(reversed(chat_reverso))

    chat = (
        [""] * (capacidade - len(chat))
        + chat
    )

    for item in chat:

        tela.append(
            "|" +
            " " +
            cortar(
                item,
                COLS - 4
            ).ljust(COLS - 4) +
            " |"
        )

    tela.append(
        "+" +
        "-" * (COLS - 2) +
        "+"
    )

    assert len(tela) == ROWS - 1

    assert all(
        len(line) == COLS
        for line in tela
    )

    return "\n".join(tela)


def desenhar(conversa):

    tela = render(conversa)

    sys.stdout.write(
        CLEAR +
        HIDE_CURSOR +
        GREEN +
        tela +
        RESET +
        "\n" +
        SHOW_CURSOR
    )

    sys.stdout.flush()


def consultar(pergunta, conversa):

    conversa_temp = conversa + [
        {
            "role": "user",
            "content": pergunta
        },
        {
            "role": "assistant",
            "content": ""
        }
    ]

    def atualizar(resposta_atual):

        conversa_temp[-1]["content"] = resposta_atual

        desenhar(conversa_temp)

    try:

        resposta = sermax_core.perguntar(
            pergunta,
            receber_parte=atualizar
        )

        conversa.extend([
            {
                "role": "user",
                "content": pergunta
            },
            {
                "role": "assistant",
                "content": resposta
            }
        ])

        return resposta

    except Exception as erro:

        resposta = "Erro no SERMAX: " + str(erro)

        conversa.extend([
            {
                "role": "user",
                "content": pergunta
            },
            {
                "role": "assistant",
                "content": resposta
            }
        ])

        return resposta

"""Camada local de reconhecimento de voz para SERMAX v2.7.2.
Validação WAV sem importar o módulo wave, que pode faltar no executável congelado.
Não baixa arquivos nem chama serviços externos.
"""
from pathlib import Path
import os
import subprocess

TERMINAL_DIR = Path(__file__).resolve().parent
SERMAX_DIR = Path(os.environ.get("SERMAX_ROOT", str(TERMINAL_DIR.parent))).resolve()
USER_DOWNLOADS = Path(os.environ.get("USERPROFILE", str(Path.home()))) / "Downloads"

def _arquivos_por_nome(raiz, nome):
    """Procura arquivos já instalados sem acessar a internet."""
    if not raiz.is_dir():
        return
    try:
        for pasta, diretorios, arquivos in os.walk(raiz):
            diretorios[:] = [d for d in diretorios if d.lower() not in {
                "__pycache__", ".git", "build", "dist", "backups", "backup"
            }]
            if nome in arquivos:
                yield Path(pasta) / nome
    except (OSError, PermissionError):
        return

def localizar_whisper_e_modelo():
    """Localiza Whisper e modelo em pastas conhecidas, sem baixar nem alterar arquivos."""
    exe_alvos = [
        SERMAX_DIR / "voz" / "whisper" / "Release" / "whisper-cli.exe",
        SERMAX_DIR / "voz" / "whisper-cli.exe",
    ]
    mod_alvos = [SERMAX_DIR / "voz" / "modelos" / "ggml-tiny.bin"]
    exe = next((p for p in exe_alvos if p.is_file()), None)
    modelo = next((p for p in mod_alvos if p.is_file()), None)
    raizes = [SERMAX_DIR, USER_DOWNLOADS]
    if not exe:
        for raiz in raizes:
            exe = next(_arquivos_por_nome(raiz, "whisper-cli.exe"), None)
            if exe: break
    if not modelo:
        for raiz in raizes:
            modelo = next(_arquivos_por_nome(raiz, "ggml-tiny.bin"), None)
            if modelo: break
    return (str(exe) if exe else None, str(modelo) if modelo else None)

def diagnostico_voz():
    exe, modelo = localizar_whisper_e_modelo()
    return {"whisper": exe, "modelo": modelo, "pronto": bool(exe and modelo)}

def _validar_wav(caminho):
    """Valida RIFF/WAVE lendo cabeçalhos, sem depender do módulo wave."""
    with open(caminho, "rb") as f:
        cab = f.read(12)
        if len(cab) < 12 or cab[0:4] != b"RIFF" or cab[8:12] != b"WAVE":
            raise ValueError("Arquivo inválido: precisa ser WAV RIFF, não basta renomear M4A para WAV.")
        formato = None
        canais = taxa = bits = None
        tem_dados = False
        while True:
            chunk = f.read(8)
            if len(chunk) < 8: break
            nome, tamanho = chunk[:4], int.from_bytes(chunk[4:8], "little")
            conteudo = f.read(min(tamanho, 64) if nome == b"fmt " else min(tamanho, 0))
            if nome == b"fmt ":
                if len(conteudo) < 16: raise ValueError("Cabeçalho WAV incompleto.")
                formato = int.from_bytes(conteudo[0:2], "little")
                canais = int.from_bytes(conteudo[2:4], "little")
                taxa = int.from_bytes(conteudo[4:8], "little")
                bits = int.from_bytes(conteudo[14:16], "little")
                f.seek(max(0, tamanho - len(conteudo)), 1)
            else:
                if nome == b"data" and tamanho > 0: tem_dados = True
                f.seek(tamanho, 1)
            if tamanho & 1: f.seek(1, 1)
            if formato is not None and tem_dados: break
    if not tem_dados or not canais or not taxa:
        raise ValueError("WAV vazio ou sem dados de áudio.")
    if formato != 1:
        raise ValueError("WAV deve usar PCM não comprimido. Converta para PCM 16-bit, mono, 16 kHz.")
    if canais not in (1, 2) or taxa < 8000 or bits not in (8, 16, 24, 32):
        raise ValueError("Parâmetros WAV incompatíveis. Recomendado: PCM 16-bit, mono, 16 kHz.")

def transcrever_arquivo(caminho_audio):
    audio = Path(caminho_audio).expanduser()
    if not audio.is_absolute(): audio = Path.cwd() / audio
    if not audio.is_file(): raise FileNotFoundError(f"Áudio não encontrado: {audio}")
    if audio.suffix.lower() != ".wav":
        raise ValueError("O arquivo precisa estar convertido para WAV; renomear M4A não converte o áudio.")
    _validar_wav(audio)
    exe, modelo = localizar_whisper_e_modelo()
    if not exe or not modelo:
        raise FileNotFoundError("Whisper/modelo não localizados. Coloque os arquivos em C:\\SERMAX\\voz conforme LEIA-ME.")
    saida = audio.with_name(audio.stem + "_sermax_transcricao")
    cmd = [exe, "-m", modelo, "-f", str(audio), "-l", "pt", "-otxt", "-of", str(saida), "-nt"]
    try:
        proc = subprocess.run(cmd, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=180, cwd=str(Path(exe).parent), shell=False)

# -*- coding: utf-8 -*-
"""Histórico local do SERMAX; arquivo ao lado do núcleo da instalação."""
import datetime
import json
import os
from pathlib import Path

ROOT = Path(os.environ.get("SERMAX_ROOT") or Path(__file__).resolve().parent).resolve()
PATH = ROOT / "historico_comandos.json"
MAX = 100
CONTROLES = ("/historico", "/repetir", "/reiniciar", "/sair", "/ajuda")


def _controle(comando):
    c = str(comando or "").strip().lower()
    return any(c == prefixo or c.startswith(prefixo + " ") for prefixo in CONTROLES)


def carregar():
    try:
        with PATH.open("r", encoding="utf-8") as f:
            d = json.load(f)
        if isinstance(d, dict) and isinstance(d.get("comandos"), list):
            d["comandos"] = [x for x in d["comandos"] if isinstance(x, dict)]
            return d
    except (OSError, UnicodeError, json.JSONDecodeError):
        pass
    return {"comandos": []}


def _arquivo_existente_valido():
    """Impede que uma gravação substitua silenciosamente um histórico inválido."""
    if not PATH.exists():
        return True
    try:
        with PATH.open("r", encoding="utf-8") as f:
            d = json.load(f)
        return isinstance(d, dict) and isinstance(d.get("comandos"), list)
    except (OSError, UnicodeError, json.JSONDecodeError):
        return False


def registrar(comando):
    comando = str(comando or "").strip()
    if not comando or _controle(comando):
        return False
    # Carregar() mantém leitura tolerante, mas gravar sobre um arquivo inválido
    # apagaria a evidência original. Recusar a escrita preserva os bytes existentes.
    if not _arquivo_existente_valido():
        return False
    d = carregar()
    itens = d["comandos"]
    # Cada entrada é registrada uma vez; não duplica imediatamente a mesma pergunta.
    if itens and str(itens[0].get("comando", "")) == comando:
        return False
    itens.insert(0, {"comando": comando, "data": datetime.datetime.now().isoformat(timespec="seconds")})
    d["comandos"] = itens[:MAX]
    tmp = PATH.with_suffix(PATH.suffix + ".tmp")
    try:
        ROOT.mkdir(parents=True, exist_ok=True)
        with tmp.open("w", encoding="utf-8", newline="\n") as f:
            json.dump(d, f, ensure_ascii=False, indent=2)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, PATH)
        return True
    except OSError:
        try:
            tmp.unlink(missing_ok=True)
        except OSError:
            pass
        return False


def listar(limite=10):
    itens = carregar()["comandos"]
    if not itens:
        return "Ainda não há comandos salvos."
    linhas = ["Histórico persistente (mais recentes primeiro):"]
    for i, x in enumerate(itens[:max(1, min(int(limite), 30))], 1):
        linhas.append(f"{i}. {x.get('comando', '')} [{x.get('data', '')}]")
    linhas.append("Use /repetir ou /repetir N.")
    return "\n".join(linhas)


def obter(indice=1):
    try:
        itens = carregar()["comandos"]
        n = int(indice)
        if 1 <= n <= len(itens):
            comando = str(itens[n - 1].get("comando", "")).strip()
            return comando or None
    except (TypeError, ValueError, KeyError):
        pass
    return None

Não invente tarefas pessoais, compromissos, horários ou prazos. Não altere a lista fornecida.

def _safe(s, n=240):
    if not isinstance(s, str): return ""
    return re.sub(r"\s+", " ", re.sub(r"[\x00-\x1f\x7f]", " ", s)).strip()[:n]

def _parse_lista(texto):
    # Accept "organizar: A, B, C", newline-separated items, or semicolons.
    partes = re.split(r"[;\n]+", texto)
    itens=[]
    for parte in partes:
        parte=parte.strip(" \t-•")
        if not parte: continue
        # Commas separate tasks only when there is no sentence-like phrase.
        if "," in parte:
            itens.extend(x.strip(" \t-•") for x in parte.split(",") if x.strip(" \t-•"))
        else:
            itens.append(parte)
    saida=[]
    vistos=set()
    for item in itens:
        item=_safe(item,180)
        chave=item.casefold()
        if item and chave not in vistos:
            vistos.add(chave); saida.append(item)
    return saida[:20]

def _plano_organizacao(objetivo):
    low=objetivo.lower().strip()
    verbos=("organizar","organize","planejar","planeje","ordenar")
    if not any(v in low for v in verbos) or not any(k in low for k in ("tarefa","tarefas","dia","rotina","lista","organizar:","organize:")):
        return None
    # Strip common task-intro prefixes but keep user content intact.
    m=re.search(r"\b(?:organizar|organize|planejar|planeje|ordenar)(?:\s+(?:as|minhas|minha|meu|meus))?(?:\s+(?:tarefas|lista|rotina|dia))?\s*:?\s*(.*)$", objetivo, re.I)
    remainder=m.group(1).strip(" :.-") if m else ""
    # "organizar minhas tarefas para hoje" has no actual task list.
    if not remainder or remainder.lower() in {"para hoje","hoje","do dia","para o dia de hoje"}:
        return ("Para organizar suas tarefas, preciso da lista real.\n"
                "Envie uma tarefa por linha ou separe-as por ponto e vírgula. "
                "Se souber, inclua prazo e urgência.\n"
                "Não vou inventar compromissos nem executar ações.")
    # Remove leading connective phrases before parsing.
    remainder=re.sub(r"^(?:para hoje|hoje|do dia de hoje)\s*[:,.-]?\s*", "", remainder, flags=re.I)
    itens=_parse_lista(remainder)
    # Avoid treating generic words as task items.
    itens=[x for x in itens if x.lower() not in {"para hoje","hoje","tarefas para hoje","minhas tarefas para hoje"}]

                   "Se informar vencimentos ou urgências, posso reordenar a lista.",

    return "\n".join(linhas)

def validar_resposta_plano(content, objetivo):
    """Valida a saída não confiável do modelo e retorna etapas permitidas."""
    if not isinstance(content, str) or len(content) > 16000:
        raise ValueError("resposta do modelo ausente ou acima de 16 KB")
    texto = content.strip()
    if texto.startswith("```"):
        texto = re.sub(r"^```(?:json)?\s*|\s*```$", "", texto, flags=re.I)
    inicio, fim = texto.find("{"), texto.rfind("}")
    if inicio < 0 or fim <= inicio:
        raise ValueError("JSON ausente")
    dados = json.loads(texto[inicio:fim + 1])
    if not isinstance(dados, dict) or not isinstance(dados.get("etapas"), list):
        raise ValueError("estrutura inválida")
    if not 1 <= len(dados["etapas"]) <= 4:
        raise ValueError("o plano deve conter de uma a quatro etapas")
    etapas = []
    for etapa in dados["etapas"]:
        if not isinstance(etapa, dict):
            raise ValueError("etapa inválida")
        ferramenta = etapa.get("ferramenta")
        if ferramenta not in FERRAMENTAS_PERMITIDAS:
            raise ValueError("ferramenta não permitida")
        titulo = _safe(etapa.get("titulo"), 160)
        verificacao = _safe(etapa.get("verificacao"), 200)
        if not titulo or not verificacao:
            raise ValueError("etapa sem título ou verificação")
        etapas.append((titulo, ferramenta, verificacao))
    return etapas
